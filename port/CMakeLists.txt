cmake_minimum_required(VERSION 3.20)

# ============================================================================
# Buffy the Vampire Slayer: Chaos Bleeds - Static Recompilation
#
#   buffy_chaos_bleeds   the recompiled game (src/, src/recomp/gen/)
#   buffy_launcher       "Buffy Launcher.exe": install from the disc image,
#                        settings, mods (launcher/)
# ============================================================================

project(buffy_chaos_bleeds C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Common definitions
add_compile_definitions(
    WIN32_LEAN_AND_MEAN
    NOMINMAX
)

# -- xboxrecomp toolkit ------------------------------------------------------
set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../xboxrecomp" CACHE PATH
    "Path to the xboxrecomp toolkit root directory")

# Builds the runtime libraries and provides the 'xboxrecomp' umbrella target.
add_subdirectory(${XBOXRECOMP_DIR} ${CMAKE_BINARY_DIR}/xboxrecomp)

# -- Recompiled game executable ----------------------------------------------
# Generated by tools/regen.sh (python -m tools.recomp); the file count
# follows --split, so they are globbed.
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen/recomp_*.c
)

set(GAME_SOURCES
    src/main.c
    src/recomp_manual.c
    src/buffy_crt.c
    src/buffy_dsound.c
    src/buffy_input.c
    src/buffy_movie.c
    src/buffy_frame.c
    src/buffy_settings.c
    src/buffy_menu.c
    src/buffy_gpuwait.c
    src/buffy_mods.c
)

add_executable(${PROJECT_NAME} WIN32
    ${GAME_SOURCES}
    ${RECOMP_GEN_SOURCES}
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp
    ${XBOXRECOMP_DIR}/include
)

if(MSVC)
    # The generated translation units are large.
    target_compile_options(${PROJECT_NAME} PRIVATE /bigobj)
endif()

target_link_libraries(${PROJECT_NAME} PRIVATE
    d3d11 dxgi dxguid xinput winmm dbghelp
    xboxrecomp
)

# -- Launcher ("Buffy Launcher.exe"): install from the disc image, settings, mods
add_executable(buffy_launcher WIN32 launcher/buffy_launcher.c)
set_target_properties(buffy_launcher PROPERTIES OUTPUT_NAME "Buffy Launcher")
target_compile_definitions(buffy_launcher PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_link_options(buffy_launcher PRIVATE /ENTRY:wWinMainCRTStartup)
endif()
